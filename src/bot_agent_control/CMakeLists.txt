cmake_minimum_required(VERSION 3.16)
project(bot_agent_control)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
add_compile_options(-Wall -Wextra -Wpedantic)
find_package(ament_cmake REQUIRED)
find_package(rclcpp REQUIRED)
find_package(bot_interfaces REQUIRED)
find_package(bot_social_force REQUIRED)
find_package(geometry_msgs REQUIRED)
find_package(visualization_msgs REQUIRED)
find_package(std_msgs REQUIRED)
find_package(tf2 REQUIRED)
find_package(tf2_ros REQUIRED)
find_package(tf2_geometry_msgs REQUIRED)
find_package(ament_index_cpp REQUIRED)
find_package(behaviortree_cpp REQUIRED)
file(GLOB controller_sources CONFIGURE_DEPENDS src/*.cpp)
list(REMOVE_ITEM controller_sources
  "${CMAKE_CURRENT_SOURCE_DIR}/src/bot_agent_loader.cpp"
  "${CMAKE_CURRENT_SOURCE_DIR}/src/test_agent_manager_node.cpp")
add_executable(${PROJECT_NAME} ${controller_sources})
target_include_directories(${PROJECT_NAME} PRIVATE include)
ament_target_dependencies(${PROJECT_NAME} rclcpp bot_interfaces bot_social_force geometry_msgs visualization_msgs std_msgs tf2 tf2_ros tf2_geometry_msgs ament_index_cpp behaviortree_cpp)
add_executable(bot_agent_loader src/bot_agent_loader.cpp)
target_include_directories(bot_agent_loader PRIVATE include)
ament_target_dependencies(bot_agent_loader rclcpp bot_interfaces tf2_geometry_msgs)
install(TARGETS ${PROJECT_NAME} bot_agent_loader DESTINATION lib/${PROJECT_NAME})
install(DIRECTORY behavior_trees DESTINATION share/${PROJECT_NAME})
ament_package()
