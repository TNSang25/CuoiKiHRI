cmake_minimum_required(VERSION 3.16)
project(human_agents)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
add_compile_options(-Wall -Wextra -Wpedantic)
find_package(ament_cmake REQUIRED)
find_package(rclcpp REQUIRED)
find_package(social_nav_msgs REQUIRED)
find_package(social_force_model REQUIRED)
find_package(geometry_msgs REQUIRED)
find_package(visualization_msgs REQUIRED)
find_package(std_msgs REQUIRED)
find_package(tf2 REQUIRED)
find_package(tf2_ros REQUIRED)
find_package(tf2_geometry_msgs REQUIRED)
find_package(ament_index_cpp REQUIRED)
find_package(behaviortree_cpp REQUIRED)
file(GLOB controller_sources CONFIGURE_DEPENDS src/*.cpp)
list(REMOVE_ITEM controller_sources "${CMAKE_CURRENT_SOURCE_DIR}/src/agent_loader.cpp")
# agent_controller: Behavior Trees + Social Force Model for every person.
add_executable(agent_controller ${controller_sources})
target_include_directories(agent_controller PRIVATE include)
ament_target_dependencies(agent_controller rclcpp social_nav_msgs social_force_model geometry_msgs visualization_msgs std_msgs tf2 tf2_ros tf2_geometry_msgs ament_index_cpp behaviortree_cpp)
# agent_loader: serves the scenario YAML (people, goals) to the controller.
add_executable(agent_loader src/agent_loader.cpp)
target_include_directories(agent_loader PRIVATE include)
ament_target_dependencies(agent_loader rclcpp social_nav_msgs tf2_geometry_msgs)
install(TARGETS agent_controller agent_loader DESTINATION lib/${PROJECT_NAME})
ament_package()
